cmake_minimum_required(VERSION 3.22)
project(alfa_pulse VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3 REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
set(PULSE_DEPS_DIR "${CMAKE_BINARY_DIR}/deps" CACHE PATH "Pinned dependency headers")
file(MAKE_DIRECTORY "${PULSE_DEPS_DIR}/nlohmann")
function(pinned_header url path hash)
  if(NOT EXISTS "${path}")
    file(DOWNLOAD "${url}" "${path}" EXPECTED_HASH "SHA256=${hash}" TLS_VERIFY ON)
  endif()
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL hash)
    message(FATAL_ERROR "Dependency checksum mismatch: ${path}")
  endif()
endfunction()
pinned_header("https://raw.githubusercontent.com/yhirose/cpp-httplib/v0.18.7/httplib.h" "${PULSE_DEPS_DIR}/httplib.h" "4770f8ea3d3fcd27b0b713c783d9b53bce3baddbd5adad13714011df96142526")
pinned_header("https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp" "${PULSE_DEPS_DIR}/nlohmann/json.hpp" "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
add_library(pulse_domain src/domain/policy.cpp)
target_include_directories(pulse_domain PUBLIC include)
add_library(pulse_application src/application/auth_service.cpp src/application/analytics_service.cpp src/application/ingestion_service.cpp)
target_include_directories(pulse_application SYSTEM PUBLIC "${PULSE_DEPS_DIR}")
target_link_libraries(pulse_application PUBLIC pulse_domain)
add_library(pulse_infrastructure src/infrastructure/postgres.cpp src/infrastructure/security.cpp src/infrastructure/model.cpp)
target_compile_definitions(pulse_infrastructure PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(pulse_infrastructure PUBLIC pulse_application OpenSSL::SSL OpenSSL::Crypto PostgreSQL::PostgreSQL Threads::Threads)
add_executable(pulse src/main.cpp src/presentation/http_server.cpp)
target_link_libraries(pulse PRIVATE pulse_infrastructure)
target_compile_options(pulse PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(pulse_tests tests/unit.cpp)
target_link_libraries(pulse_tests PRIVATE pulse_infrastructure)
add_test(NAME unit COMMAND pulse_tests)
